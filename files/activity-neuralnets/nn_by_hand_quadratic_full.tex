\section*{Problem Setup (shared)}
We fit a tiny neural network with a single hidden layer (2 units) and ReLU activation to the target $y=x^2+2x$ using full-batch gradient descent.

\noindent\textbf{Network:}
\begin{align*}
&\text{Hidden (2 units):} && z_1 = W_1 x + b_1,\quad h = \relu(z_1)\\
&\text{Output (scalar):} && \hat y = W_2 h + b_2
\end{align*}

\noindent\textbf{Parameters:}
$W_1=\begin{bmatrix}w_{11}\\w_{21}\end{bmatrix},\ 
b_1=\begin{bmatrix}b_{11}\\b_{21}\end{bmatrix},\ 
W_2=\begin{bmatrix}w_{2,1} & w_{2,2}\end{bmatrix},\ 
b_2\in\mathbb{R}.$

\noindent\textbf{Loss:}
$L=\tfrac12\sum_s (y_s-\hat y_s)^2.$ We take $\relu'(0)=0$.
